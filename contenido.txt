Crea un apartado en la web /horario/ para mostrar el horario del centro educativo. Todos los módulos se tienen que cargar desde la BBDD.
Tienes que guardar una relación de:

Nombre del módulo
Profesor
Siglas del módulo
Curso
Letra
color
Los profesores y alumnos, tendrán en común una tabla de personas.
Cada módulo tiene que tener un color único. 

Piensa en la tablas de horarios que es diferente a la de módulos. 

-- 1. TABLA PERSONAS 
-- Aquí se guardan tanto alumnos como profesores. 
-- Se usa un campo 'tipo' o 'rol' para diferenciarlos.
CREATE TABLE personas (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    nombre      VARCHAR(100) NOT NULL,
    apellidos   VARCHAR(100) NOT NULL,
    email       VARCHAR(150) UNIQUE,
    tipo        CHAR(01) NOT NULL,        --[P]rofesor, [A]lumno
    -- Datos específicos pueden ir aquí o en tablas satélite si fuera muy complejo
    -- Para este ejercicio, basta con esto.
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP


    --OTROS DATOS QUE ESTIMES NECESARIOS PARA EL FRAMEWORK
);

-- 2. TABLA CURSOS
CREATE TABLE cursos (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    nombre_grado    VARCHAR(50) NOT NULL,           -- Ej: "DAW", "DAM"
    curso_numero    INT NOT NULL,                   -- Ej: 1, 2
    letra           CHAR(1) NOT NULL,               -- Ej: 'A', 'B' (Requisito explícito)
    -- Concatenado en vista sería: "1 DAW A"
    UNIQUE(nombre_grado, curso_numero, letra)       -- Evita duplicados como dos "1 DAW A"

    --OTROS DATOS QUE ESTIMES NECESARIOS PARA EL FRAMEWORK
);

-- 3. TABLA MÓDULOS 
CREATE TABLE modulos (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    nombre          VARCHAR(100) NOT NULL,              -- Ej: "Desarrollo Entorno Servidor"
    siglas          VARCHAR(10) NOT NULL,               -- Ej: "DWES" (Requisito explícito)
    color           VARCHAR(7) NOT NULL UNIQUE,         -- Ej: "#FF5733" (Requisito: Color único)
    curso_asignado  INT NOT NULL,                       -- FK opcional si un módulo pertenece estrictamente a un curso
    FOREIGN KEY (curso_asignado) REFERENCES cursos(id)

    --OTROS DATOS QUE ESTIMES NECESARIOS PARA EL FRAMEWORK
);

-- 5. TABLA aulas
CREATE TABLE aulas(
    id      INT AUTO_INCREMENT,
    nombre  VARCHAR(100) NOT NULL,
    letra   CHAR(1) NOT NULL,
    numero  INT NOT NULL UNIQUE,
    planta  CHAR(1), -- [P]rimera, [S]egunda, [T]ercera

    --OTROS DATOS QUE ESTIMES NECESARIOS PARA EL FRAMEWORK
)

-- 5. TABLA HORARIOS 
-- Es diferente a la de módulos porque define CUÁNDO ocurre la clase.
CREATE TABLE horarios (
    id                  INT AUTO_INCREMENT PRIMARY KEY,
    dia                 CHAR(01) NOT NULL,                -- [L]unes, [M]artes, [M]iercoles, [J]ueves, [V]iernes NOT NULL,
    hora_inicio         TIME NOT NULL,
    hora_fin            TIME NOT NULL,
    
    -- RELACIONES
    id_modulo INT NOT NULL,      -- Qué se imparte (y de aquí sacamos el color y siglas)
    id_profesor INT NOT NULL,    -- Quién lo imparte (FK a tabla PERSONAS)
    id_aula INT NULL,            -- Dónde (Reciclamos la tabla Aulas del ejercicio anterior)
    
    FOREIGN KEY (id_modulo) REFERENCES modulos(id),
    FOREIGN KEY (id_profesor) REFERENCES personas(id),
    -- Validación extra: Asegurarse por código que la persona sea 'Profesor'
    FOREIGN KEY (id_aula) REFERENCES aulas(id) 



    --OTROS DATOS QUE ESTIMES NECESARIOS PARA EL FRAMEWORK
);




Esta es la query que debes pegar en tu código PHP para rellenar la tabla del horario:
  SELECT 
    h.dia,
    h.hora_inicio,
    h.hora_fin,
    m.nombre AS nombre_modulo,
    m.siglas AS siglas_modulo,
    m.color AS color_modulo,
    CONCAT(p.nombre, ' ', p.apellidos) AS nombre_profesor,
    c.curso_numero,
    c.nombre_grado,
    c.letra
FROM 
    horarios h
    JOIN modulos m ON h.id_modulo = m.id_modulo
    JOIN personas p ON h.id_profesor = p.id_persona
    JOIN cursos c ON m.curso_asignado = c.id_curso
ORDER BY 
    h.dia, h.hora_inicio;
